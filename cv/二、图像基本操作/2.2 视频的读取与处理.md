### 2.2.1 图像属性
``` python
import cv2
import matplotlib.pyplot as plt
import numpy as np

# shape
img = cv2.imread('2/img/mao.jpg')
print(img.shape) # (148, 245, 3) hwc

# 灰度图
img = cv2.imread('2/img/mao.jpg', cv2.IMREAD_GRAYSCALE)
print(img)
# [[211 211 211 ... 174 168 165]
#  [211 211 211 ... 180 177 175]
#  [211 211 210 ... 189 188 187]
#  ...

print(img.shape) # (148, 245) hw
cv2.imshow('GrayImg', img)
cv2.waitKey(0)
cv2.destroyAllWindows()

# 保存
cv2.imwrite('2/img/mao_gray.jpg', img)

# 打印类型
print(type(img)) # <class 'numpy.ndarray'>

# 图片大小(像素个数) 
print(img.size) # 36260

# 存储类型
print(img.dtype) # uint8
```

### 2.2.2 视频操作
``` python
import cv2
import matplotlib.pyplot as plt
import numpy as np

vc = cv2.VideoCapture('2/video/test_video.mp4')
if vc.isOpened(): # 判断文件是否打开成功
    while True:
        ret, frame = vc.read() # 读取帧
        if frame is None: # 帧为 None 时说明视频到结尾了
            break
        if ret == True: # 帧读取成功
            gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY) # 将帧数据转为灰度图 BGR2GRAY
            cv2.imshow('result', gray) # 展示当前帧
            if cv2.waitKey(10) & 0xFF==27: # 等待10ms并且当按下 ESC 时播放中止
                break
    
vc.release()
cv2.destroyAllWindows()
```
