### 2.3.1 感兴趣区域裁剪及通道过滤
``` python
import cv2
import matplotlib.pyplot as plt
import numpy as np

def cv_show(window_name, img):
    cv2.imshow(window_name, img)

img = cv2.imread('2/img/mao.jpg')
cat = img[0:50, 0:100] # 选择感兴趣的像素点 hw
cv_show('cat', cat)

# cat
b, g, r = cv2.split(cat) # b g r通道拆分
print(b)
print(b.shape) # (50, 100)
print(g.shape) # (50, 100)

# img
b, g, r = cv2.split(img)
print(b.shape) # (148, 245)
print(g.shape) # (148, 245)
img = cv2.merge((b, g, r)) # 通道合并
print(img.shape) # (148, 245, 3) hwc

# 只保留R
cur_img = img.copy()
cur_img[:, :, 0] = 0 # B 通道的值置0
cur_img[:, :, 1] = 0 # G 通道的值置0
cv_show('R', cur_img)

cv2.waitKey(0)
cv2.destroyAllWindows()
```
