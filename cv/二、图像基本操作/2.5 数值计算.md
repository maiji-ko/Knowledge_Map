### 2.5.1 像素计算及图片融合
``` python
import cv2
import matplotlib.pyplot as plt
import numpy as np

img_cat = cv2.imread('2/img/mao.jpg')
img_dog = cv2.imread('2/img/dog.jpg')
print(img_cat[:5, :, 0]) # 输出5行 B通道
# [[227 227 227 ... 197 191 188]
#  [227 227 227 ... 203 198 198]
#  [227 227 226 ... 210 207 208]
#  [226 226 226 ... 218 217 219]
#  [226 226 225 ... 229 228 231]]

# 所有值 +10
img_cat2 = img_cat + 10
print(img_cat2[:5, :, 0]) # 输出5行 B通道
# [[237 237 237 ... 207 201 198]
#  [237 237 237 ... 213 208 208]
#  [237 237 236 ... 220 217 218]
#  [236 236 236 ... 228 227 229]
#  [236 236 235 ... 239 238 241]]

# 两张大小相同的图像相加
print((img_cat + img_cat2)[:5, :, 0]) # 数值溢出 (237 + 247) % 256 = 208
# [[208 208 208 ... 148 136 130]
#  [208 208 208 ... 160 150 150]
#  [208 208 206 ... 174 168 170]
#  [206 206 206 ... 190 188 192]
#  [206 206 204 ... 212 210 216]]

# cv 提供函数，溢出取最大值
print(cv2.add(img_cat, img_cat2)[:5, :, 0])
# [[255 255 255 ... 255 255 255]
#  [255 255 255 ... 255 255 255]
#  [255 255 255 ... 255 255 255]
#  [255 255 255 ... 255 255 255]
#  [255 255 255 ... 255 255 255]]

# 图像融合
print(img_cat.shape) # (148, 245, 3) hwc
print(img_dog.shape) # (148, 148, 3) 图像大小不一样直接执行 + 法操作报错

# 图像大小变换
img_dog = cv2.resize(img_dog, (245, 148)) # 图像缩放到 (245, 148) wh  -- cat
print(img_dog.shape) # (148, 245, 3)

img_cat3 = cv2.resize(img_cat, (0, 0), fx = 3, fy = 1) # 长变为三倍 高不变
print(img_cat3.shape) # (148, 735, 3) hwc

# 融合
res = cv2.addWeighted(img_cat, 0.4, img_dog, 0.6, 0) # 亮度占比 cat 0.4 dog 0.6
plt.subplot(131), plt.imshow(img_cat, 'gray'), plt.title('CAT')
plt.subplot(132), plt.imshow(img_dog, 'gray'), plt.title('DOG')
plt.subplot(133), plt.imshow(res, 'gray'), plt.title('RES')
plt.show()
```
